// Coloreado por lectura individual (umbral domiciliario ESH 2023)
function classifyReading(sbp, dbp) {
  if (sbp === null || dbp === null || isNaN(sbp) || isNaN(dbp)) return null;
  if (sbp > 180 || dbp > 120) return BP_CATS.crisis;
  if (sbp < 90  || dbp < 60 ) return BP_CATS.low;
  if (sbp >= 160 || dbp >= 100) return BP_CATS.g2;
  if (sbp >= HOME_THRESHOLD.sbp || dbp >= HOME_THRESHOLD.dbp) return BP_CATS.g1;
  if (sbp >= 120 || dbp >= 80) return BP_CATS.elev;
  return BP_CATS.normal;
}

function fmt(v, d = 0) {
  return (v === null || v === undefined || isNaN(v)) ? '—' : Number(v).toFixed(d);
}
function mean(arr) {
  const v = arr.filter(x => x !== null && !isNaN(x));
  if (!v.length) return null;
  return v.reduce((a, b) => a + b, 0) / v.length;
}
function sd(arr) {
  const v = arr.filter(x => x !== null && !isNaN(x));
  if (v.length < 2) return null;
  const m = mean(v);
  return Math.sqrt(v.reduce((acc, x) => acc + (x - m) ** 2, 0) / (v.length - 1));
}

function makeBPCell(day, period, isSbp = true) {
  const td = document.createElement('td');
  td.style.textAlign = 'center';
  const input = document.createElement('input');
  input.type = 'number';
  input.min = isSbp ? '60' : '40';
  input.max = isSbp ? '280' : '160';
  input.id = (isSbp ? 'sbp' : 'dbp') + `_${day}_${period}`;
  input.dataset.day = day; input.dataset.period = period; input.dataset.type = isSbp ? 'sbp' : 'dbp';
  input.className = 'bp-input'; input.placeholder = '—';
  td.appendChild(input);
  return td;
}

function attachInputListeners() {
  document.querySelectorAll('.bp-input').forEach(el => {
    el.addEventListener('input', onBPInput);
  });
}

function attachAutoAdvance() {
  const DAYS = getDays();
  const order = [];
  for (let day = 0; day < DAYS; day++) {
    for (const period of PERIODS) {
      order.push(`sbp_${day}_${period}`);
      order.push(`dbp_${day}_${period}`);
      order.push(`fc_${day}_${period}`);
    }
  }
  order.forEach((id, idx) => {
    const el = document.getElementById(id);
    if (!el) return;
    const advance = () => {
      const next = document.getElementById(order[idx + 1]);
      if (next) { next.focus(); next.select(); }
    };
    el.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') { e.preventDefault(); advance(); }
    });
    el.addEventListener('input', () => {
      const val = el.value;
      if (!val) return;
      const isSbp = id.startsWith('sbp');
      if (isSbp && val.length >= 3) advance();
      else if (!isSbp && val.length >= 2 && parseInt(val) >= 30) advance();
    });
  });
}

function onBPInput(e) {
  const { day, period } = e.target.dataset;
  const sbpEl = document.getElementById(`sbp_${day}_${period}`);
  const dbpEl = document.getElementById(`dbp_${day}_${period}`);
  const sbp = parseFloat(sbpEl?.value);
  const dbp = parseFloat(dbpEl?.value);
  const cls = ['bg-low','bg-ok','bg-elev','bg-s1','bg-s2','bg-cri','inv-error'];
  if (sbpEl) sbpEl.classList.remove(...cls);
  if (dbpEl) dbpEl.classList.remove(...cls);
  if (!isNaN(sbp) && !isNaN(dbp)) {
    if (dbp >= sbp) {
      sbpEl?.classList.add('inv-error');
      dbpEl?.classList.add('inv-error');
    } else {
      const col = classifyReading(sbp, dbp);
      if (col) { sbpEl?.classList.add(col.inputCls); dbpEl?.classList.add(col.inputCls); }
    }
  }
  const pamEl = document.getElementById(`pam_${day}_${period}`);
  if (pamEl) {
    if (!isNaN(sbp) && !isNaN(dbp) && dbp < sbp) {
      pamEl.textContent = Math.round(dbp + (sbp - dbp) / 3);
    } else {
      pamEl.textContent = '—';
    }
  }
  updateDayAvg(parseInt(day));
}

function updateDayAvg(day) {
  const vals = { sbp:[], dbp:[], fc:[] };
  PERIODS.forEach(p => {
    const s = parseFloat(document.getElementById(`sbp_${day}_${p}`)?.value);
    const d = parseFloat(document.getElementById(`dbp_${day}_${p}`)?.value);
    const f = parseFloat(document.getElementById(`fc_${day}_${p}`)?.value);
    if (!isNaN(s) && !isNaN(d) && d < s && s >= 60 && d >= 40) {
      vals.sbp.push(s); vals.dbp.push(d);
    }
    if (!isNaN(f)) vals.fc.push(f);
  });
  const avg = (arr) => arr.length ? (arr.reduce((a,b)=>a+b,0)/arr.length) : null;
  const avgSbp = avg(vals.sbp), avgDbp = avg(vals.dbp), avgFc = avg(vals.fc);
  const avgPam = (avgSbp !== null && avgDbp !== null) ? (avgDbp + (avgSbp - avgDbp)/3) : null;
  const sbpEl = document.getElementById(`avg_sbp_${day}`);
  const dbpEl = document.getElementById(`avg_dbp_${day}`);
  const fcEl  = document.getElementById(`avg_fc_${day}`);
  const pamEl = document.getElementById(`avg_pam_${day}`);
  if (sbpEl) sbpEl.textContent = avgSbp !== null ? Math.round(avgSbp) : '—';
  if (dbpEl) dbpEl.textContent = avgDbp !== null ? Math.round(avgDbp) : '—';
  if (fcEl)  fcEl.textContent  = avgFc  !== null ? Math.round(avgFc)  : '—';
  if (pamEl) pamEl.textContent = avgPam !== null ? Math.round(avgPam) : '—';
}

/* ============================================================
   GRID READER
============================================================ */
function readGrid() {
  const DAYS = getDays();
  const rows = [];
  for (let day = 0; day < DAYS; day++) {
    const row = { day };
    PERIODS.forEach(p => {
      const sbpVal = parseFloat(document.getElementById(`sbp_${day}_${p}`)?.value);
      const dbpVal = parseFloat(document.getElementById(`dbp_${day}_${p}`)?.value);
      const fcVal  = parseFloat(document.getElementById(`fc_${day}_${p}`)?.value);
      row[p] = {
        sbp: isNaN(sbpVal) ? null : sbpVal,
        dbp: isNaN(dbpVal) ? null : dbpVal,
        fc:  isNaN(fcVal)  ? null : fcVal,
        valid:     !isNaN(sbpVal) && !isNaN(dbpVal),
        physValid: !isNaN(sbpVal) && !isNaN(dbpVal) && dbpVal < sbpVal && sbpVal >= 60 && dbpVal >= 40
      };
    });
    rows.push(row);
  }
  return rows;
}

/* ============================================================
   CALCULATE
============================================================ */
function calculate() {
  const rows = readGrid();
  const DAYS = getDays();
  const daysWithData = rows.filter(row => PERIODS.some(p => row[p].valid)).length;
  const profileKey = getProfile();
  const target = TARGETS[profileKey] || TARGETS.nom_general;

  const byPeriod = { manana: { sbp:[], dbp:[], fc:[] }, tarde:{ sbp:[], dbp:[], fc:[] }, noche:{ sbp:[], dbp:[], fc:[] } };
  const allSbp = [], allDbp = [], allFc = [];
  let clsCnt = { crisis:0, g2:0, g1:0, elev:0, normal:0, low:0 };
  let tgtOk = 0, tgtAbove = 0, tgtLow = 0;
  let totalN = 0, validN = 0;

  rows.forEach(row => {
    PERIODS.forEach(p => {
      totalN++;
      const r = row[p];
      if (!r.valid) return;
      validN++;
      if (!r.physValid) return;
      byPeriod[p].sbp.push(r.sbp);
      byPeriod[p].dbp.push(r.dbp);
      if (r.fc !== null) { byPeriod[p].fc.push(r.fc); allFc.push(r.fc); }
      allSbp.push(r.sbp); allDbp.push(r.dbp);
      const c = classifyReading(r.sbp, r.dbp);
      if (c) clsCnt[c.level] = (clsCnt[c.level] || 0) + 1;
      const isHypo = r.sbp < 90 || r.dbp < 60;
      if (isHypo) { tgtLow++; }
      else if (r.sbp < target.sbp && r.dbp < target.dbp) { tgtOk++; }
      else { tgtAbove++; }
    });
  });

  const periodAvg = {};
  PERIODS.forEach(p => {
    const mSbp = mean(byPeriod[p].sbp), mDbp = mean(byPeriod[p].dbp);
    periodAvg[p] = {
      sbp: mSbp, dbp: mDbp,
      fc:  mean(byPeriod[p].fc),
      n:   byPeriod[p].sbp.length,
      pam: (mSbp !== null && mDbp !== null) ? mDbp + (mSbp - mDbp)/3 : null
    };
  });

  const overallSbp = mean(allSbp);
  const overallDbp = mean(allDbp);
  const overallPam = (overallSbp !== null && overallDbp !== null) ? overallDbp + (overallSbp - overallDbp)/3 : null;

  const morn = periodAvg['manana'];
  const aft  = periodAvg['tarde'];
  const nigh = periodAvg['noche'];
  const surgeSbp = (morn.sbp !== null && nigh.sbp !== null) ? morn.sbp - nigh.sbp : null;

  let dipperPct = null, dipperStatus = null;
  const dayVals = [morn.sbp, aft.sbp].filter(x => x !== null);
  const dayAvg = dayVals.length ? dayVals.reduce((a,b)=>a+b,0)/dayVals.length : null;
  if (dayAvg !== null && nigh.sbp !== null && dayAvg > 0) {
    dipperPct = (dayAvg - nigh.sbp) / dayAvg * 100;
    if (dipperPct > 10) dipperStatus = 'dipper';
    else               dipperStatus = 'non-dipper'; // incluye <0: insuficiente resolución para "reverse" sin MAPA
  }

  const sdSbp = sd(allSbp);
  const overallFc = mean(allFc);
  const sdFc  = sd(allFc);

  const bpLoad = validN > 0 ? (allSbp.filter((s,i) => s >= target.sbp || allDbp[i] >= target.dbp).length / validN * 100) : 0;

  // ── Sesgos ──────────────────────────────────────────────────────
  const biasFlags = [];
  let fcMissingCount = 0;
  let invalidEntries = [];
  let partialDays = [];
  let emptyDays = [];

  rows.forEach(row => {
    const entered = PERIODS.filter(p => row[p].valid);
    const physOk  = PERIODS.filter(p => row[p].physValid);
    if (entered.length === 0) {
      emptyDays.push(row.day + 1);
    } else if (physOk.length < entered.length) {
      PERIODS.forEach(p => {
        if (row[p].valid && !row[p].physValid) {
          invalidEntries.push({ day: row.day+1, period: PERIOD_META[p].short,
            sbp: row[p].sbp, dbp: row[p].dbp });
        }
      });
    }
    if (entered.length > 0 && entered.length < 3) {
      partialDays.push({ day: row.day+1, n: physOk.length, entered: entered.length });
    }
    PERIODS.forEach(p => {
      if (row[p].physValid && row[p].fc === null) fcMissingCount++;
    });
  });

  if (invalidEntries.length > 0) {
    biasFlags.push({ sev:'error', msg:
      `<strong>Error de entrada (DIA ≥ SIS):</strong> ${invalidEntries.map(e =>
        `D${e.day} ${e.period} (${e.sbp}/${e.dbp})`).join(', ')}.
       Lecturas excluidas del cálculo. Verificar si hubo inversión de valores al registrar.` });
  }
  if (emptyDays.length > 0) {
    biasFlags.push({ sev:'warn', msg:
      `<strong>Días sin registro:</strong> D${emptyDays.join(', D')}.
       Los promedios y la carga tensional no representan esos días.
       Si la TA fue diferente (mejor adherencia, estrés, etc.) el promedio global puede estar sesgado.` });
  }
  if (partialDays.length > 0) {
    biasFlags.push({ sev:'warn', msg:
      `<strong>Registro incompleto:</strong> ${partialDays.map(d =>
        `D${d.day} (${d.n}/3 registros válidos)`).join(', ')}.
       El promedio diario de esos días se calcula con &lt; 3 horarios — puede no representar el patrón circadiano completo.` });
  }
  if (fcMissingCount > 0) {
    biasFlags.push({ sev:'info', msg:
      `<strong>FC no registrada en ${fcMissingCount} lectura${fcMissingCount>1?'s':''}.</strong>
       No se puede evaluar respuesta cronotrópica, variabilidad de FC ni señales de arritmia en esos registros.` });
  }
  if (validN < 12) {
    biasFlags.push({ sev:'warn', msg:
      `<strong>Lecturas insuficientes (n=${validN}).</strong>
       El mínimo recomendado para decisión clínica es ≥12 lecturas (AHA/ACC). Los resultados tienen menor confiabilidad estadística.` });
  }
  const extremeN = allSbp.filter(v => v >= 250).length;
  if (extremeN > 0) {
    biasFlags.push({ sev:'warn', msg:
      `<strong>${extremeN} lectura${extremeN>1?'s':''} ≥250 mmHg.</strong>
       Verificar técnica antes de tomar decisión terapéutica. En crisis real el paciente requiere evaluación presencial urgente.` });
  }
  const last2empty = rows.slice(-2).every(r => !PERIODS.some(p => r[p].valid));
  if (last2empty && emptyDays.length > 0) {
    biasFlags.push({ sev:'info', msg:
      `<strong>Posible sesgo de abandono:</strong> los últimos días del registro están vacíos.
       El control en el cierre del período es desconocido. Considerar repetir registro si la decisión es de alto impacto.` });
  }
  if (validN > 0 && emptyDays.length === 0 && partialDays.length === 0) {
    biasFlags.push({ sev:'ok', msg:
      `Registro completo: sin supuestos de imputación. Los promedios reflejan todas las lecturas registradas.` });
  }

  const payload = {
    overallSbp, overallDbp, overallPam, clsCnt, tgtOk, tgtAbove, tgtLow,
    validN, totalN, DAYS, daysWithData,
    periodAvg, target, profileKey,
    surgeSbp, dipperPct, dipperStatus,
    sdSbp, bpLoad,
    overallFc, sdFc,
    allRowsHaveData: rows.map(row => PERIODS.some(p => row[p].valid)),
    biasFlags
  };

  document.getElementById('results-section').style.display = '';
  renderDualClass(payload);
  renderBanner(payload);
  renderQuality(payload);
  renderPeriods(payload);
  renderTTR(payload);
  renderPattern(payload);
  renderBias(payload);
  renderInterpretation(payload);
  document.getElementById('results-section')?.scrollIntoView?.({ behavior: 'smooth' });
}

    dipNote = `Descenso nocturno ${fmt(dipperPct, 1)}% · Patrón protector (>10%)`;
  } else if (dipperStatus === 'non-dipper') {
    dipLabel = '⚠️ Non-Dipper'; dipColor = '#f97316';
    dipNote = `Descenso nocturno ${fmt(dipperPct, 1)}% · Riesgo CV elevado (<10%) · Confirmar con MAPA 24h`;
  }

  const ppItems = PERIODS.map(prd => {
    const pa = periodAvg[prd];
    const pp = (pa.sbp !== null && pa.dbp !== null) ? pa.sbp - pa.dbp : null;
    const ppColor = pp !== null ? (pp > 60 ? '#ef4444' : pp < 30 ? '#3b82f6' : '#22c55e') : '#94a3b8';
    return `<div style="text-align:center;">
      <div style="font-size:11px;color:var(--c-muted);">${PERIOD_META[prd].icon}</div>
      <div style="font-size:20px;font-weight:800;color:${ppColor};">${pp !== null ? fmt(pp,0) : '—'}</div>
      <div style="font-size:10px;color:var(--c-muted);">mmHg</div>
    </div>`;
  }).join('');

  const mornSbp  = periodAvg['manana'].sbp;
  const nightSbp = periodAvg['noche'].sbp;
  const aftSbp   = periodAvg['tarde'].sbp;
  const gradMN = (mornSbp !== null && nightSbp !== null) ? `${fmt(mornSbp,0)} → ${fmt(nightSbp,0)} mmHg` : '—';
  const gradMA = (mornSbp !== null && aftSbp !== null)   ? `${fmt(mornSbp,0)} → ${fmt(aftSbp,0)} mmHg` : '—';

  el.innerHTML = `
    <div class="pat-grid">
      <div class="pat-card">
        <div class="pat-label">Surge Matutino (SBP)</div>
        <div class="pat-value" style="color:${surgeColor};">${surgeSbp !== null ? (surgeSbp >= 0 ? '+' : '') + fmt(surgeSbp,1) + ' mmHg' : '—'}</div>
        <div class="pat-note">${surgeNote}</div>
      </div>
      <div class="pat-card">
        <div class="pat-label">Patrón Dipper Nocturno</div>
        <div class="pat-value" style="color:${dipColor};font-size:14px;">${dipLabel}</div>
        <div class="pat-note">${dipNote}</div>
      </div>
      <div class="pat-card">
        <div class="pat-label">Gradiente Mañana → Noche</div>
        <div class="pat-value" style="color:var(--c-primary);font-size:15px;">${gradMN}</div>
        <div class="pat-note">Promedio SBP por horario</div>
      </div>
      <div class="pat-card">
        <div class="pat-label">Gradiente Mañana → Tarde</div>
        <div class="pat-value" style="color:var(--c-primary);font-size:15px;">${gradMA}</div>
        <div class="pat-note">Efecto del fármaco en la jornada (vida media)</div>
      </div>
      <div class="pat-card" style="grid-column:span 2;">
        <div class="pat-label">Presión de Pulso por Horario</div>
        <div style="display:flex;gap:20px;margin-top:8px;">${ppItems}</div>
        <div class="pat-note" style="margin-top:6px;">PP normal 40–60 mmHg · PP >60: rigidez arterial, riesgo CV aumentado (ESH 2023)</div>
      </div>
    </div>
    <div class="pat-disclaimer">
      ⚠️ <strong>Limitación metodológica:</strong> el patrón dipper/non-dipper y el surge matutino se estiman aquí como
      <em>aproximación</em> con 3 puntos domiciliarios (mañana/tarde/noche). El estándar de referencia es MAPA ambulatoria de 24h
      con &gt;48 mediciones. Si el patrón non-dipper cambia una decisión clínica (p. ej., reposicionamiento
      de fármacos), solicitar MAPA 24h para confirmación antes de actuar.
    </div>
  `;
}

/* ============================================================
   RENDER: INTERPRETATION
============================================================ */
function renderInterpretation(p) {
  const { overallSbp, overallDbp, target, tgtOk,
          validN, surgeSbp, dipperStatus, dipperPct, bpLoad, sdSbp, periodAvg,
          overallFc, sdFc } = p;
  const el   = document.getElementById('sect-interpretation');
  const alEl = document.getElementById('sect-alerts');
  if (!el || !alEl) return;

  const c = classifyReading(overallSbp, overallDbp);
  if (!c) { el.innerHTML='<p style="color:var(--c-muted)">Sin datos suficientes.</p>'; return; }

  const points = [];
  const alerts  = [];

  // --- CONTROL STATUS ---
  switch (c.level) {
    case 'normal':
      points.push(`✅ TA <strong>controlada</strong> (${fmt(overallSbp,0)}/${fmt(overallDbp,0)} mmHg).
        Dentro de meta ${target.sbp}/${target.dbp}. Mantener esquema actual, reforzar adherencia y estilo de vida.
        Revisar en 3–6 meses según NOM-030.`); break;
    case 'elev':
      points.push(`🟡 TA en <strong>rango elevado</strong> (${fmt(overallSbp,0)}/${fmt(overallDbp,0)} mmHg).
        Evaluar: adherencia, restricción de sal (&lt;5g NaCl/día), actividad física, estrés, AINE/descongestionantes.
        Considerar modificación de estilo de vida antes de ajuste farmacológico si presentación reciente. NOM-030.`); break;
    case 'g1':
      points.push(`🟠 <strong>HTA Grado 1 / Etapa 1</strong> (${fmt(overallSbp,0)}/${fmt(overallDbp,0)} mmHg).
        <em>Antes de intensificar:</em> verificar técnica de medición (brazalete adecuado, reposo 5 min, brazo correcto),
        adherencia al esquema actual, consumo de AINE/descongestionantes/alcohol/sal, apnea del sueño, dolor, ansiedad.
        Si factores optimizados y persiste fuera de meta en 4–8 semanas: valorar intensificación conforme a
        cuadro básico institucional y comorbilidades (GPC-CENETEC). AHA/ACC 2025: iniciar/intensificar si riesgo CV
        ≥7.5% (ecuaciones PREVENT) o daño a órgano blanco.`); break;
    case 'g2':
      points.push(`🔴 <strong>HTA Grado 2 / Etapa 2</strong> (${fmt(overallSbp,0)}/${fmt(overallDbp,0)} mmHg).
        Requiere evaluación de daño a órgano blanco: ECG, creatinina/TFG, microalbuminuria.
        GPC-CENETEC: combinación de inicio en HTA Grado 2. Combinación preferida: IECA/ARA-II + CCB (amlodipino)
        ± diurético tiazídico. En DM: IECA/ARA-II de primera línea (nefroprotección). Siempre verificar
        adherencia y factores secundarios antes de escalar. ESH 2023.`); break;
    case 'crisis':
      alerts.push({ type:'crisis', msg:`🆘 <strong>CRISIS HIPERTENSIVA</strong> en promedio del registro
        (${fmt(overallSbp,0)}/${fmt(overallDbp,0)} mmHg). Diferenciar urgencia vs. emergencia:
        evaluar síntomas de daño agudo a órgano blanco (cefalea intensa, déficit neurológico, dolor torácico,
        disnea, alteración visual, oliguria). <br><strong>Emergencia</strong> → derivación inmediata (urgencias hospitalarias).
        <strong>Urgencia</strong> → intensificación oral + vigilancia 24–72h + seguimiento estrecho.` }); break;
    case 'low':
      alerts.push({ type:'info', msg:`🔵 Promedio en <strong>hipotensión</strong>
        (${fmt(overallSbp,0)}/${fmt(overallDbp,0)} mmHg). Evaluar: sobretratamiento,
        deshidratación, interacciones (diurético + IECA). Considerar: reducción de dosis, cambio de horario
        o eliminación de agente redundante. Descartar hipotensión ortostática (medir en decúbito/bipedestación).` }); break;
  }

  // --- CLASES DE FÁRMACOS (cualquier nivel con HTA) ---
  if (['g1','g2','crisis'].includes(c.level)) {
    points.push(`💊 <strong>Clases de primera línea:</strong> <abbr title="Inhibidor de la Enzima Convertidora de Angiotensina">IECA</abbr>/<abbr title="Antagonista del Receptor de Angiotensina II">ARA-II</abbr>, <abbr title="Calcioantagonista">CCB</abbr> (amlodipino) o diurético tiazídico/similar-tiazídico, según comorbilidad — IECA/ARA-II preferido en DM o ERC con albuminuria; CCB o tiazídico son alternativas de primera línea igualmente válidas sin esas comorbilidades.`);
  }

  // --- DISTRIBUCIÓN Y CARGA TENSIONAL ---
  const pctEnMeta = validN > 0 ? (tgtOk / validN * 100) : 0;
  points.push(`📊 <strong>${pctEnMeta.toFixed(0)}% de lecturas en meta</strong>
    (${tgtOk} de ${validN} lecturas &lt;${target.sbp}/${target.dbp} mmHg · ${target.source}).
    ${pctEnMeta < 50
      ? 'La mayoría de lecturas están fuera de meta — sugiere control insuficiente sostenido, no solo episodios aislados.'
      : pctEnMeta < 75
        ? 'Control parcial: más de la mitad en meta, pero exposición hemodinámica fuera de rango persiste.'
        : 'Buen control: mayoría de lecturas dentro de meta.'}
    Verificar adherencia y técnica de medición antes de ajustar tratamiento (NOM-030-SSA2, AHA/ACC 2025).`);

  if (bpLoad > 40) {
    points.push(`🔴 <strong>Carga tensional elevada (${fmt(bpLoad,1)}%)</strong>.
      Una carga sostenida &gt;40% se ha relacionado en estudios de MAPA con mayor daño a órgano blanco (hipertrofia
      ventricular, albuminuria) independientemente del promedio. En HBPM de 3 tomas/día interprétese como marcador
      exploratorio — no equivale a un hallazgo de MAPA — pero indica exposición hemodinámica repetida que el
      promedio puede enmascarar si hay lecturas muy dispersas.`);
  } else if (bpLoad > 25) {
    points.push(`🟡 Carga tensional moderada (${fmt(bpLoad,1)}%). Vigilar evolución. Reforzar medidas no farmacológicas.`);
  }

  // --- DIPPING ---
  if (dipperStatus === 'non-dipper') {
    points.push(`🌙 <strong>Patrón <abbr title="Descenso nocturno de PA <10% respecto al promedio diurno">Non-Dipper</abbr></strong> (descenso nocturno ${fmt(dipperPct,1)}%, &lt;10%).
      Riesgo <abbr title="Cardiovascular">CV</abbr> aumentado: <abbr title="Hipertrofia Ventricular Izquierda">HVI</abbr>, nefropatía, eventos independientes del promedio diurno.
      Confirmar con <abbr title="Monitoreo Ambulatorio de Presión Arterial">MAPA</abbr> 24h antes de modificar esquema.`);
  } else if (dipperStatus === 'dipper') {
    points.push(`🌙 Patrón <strong>Dipper normal</strong> (descenso nocturno ${fmt(dipperPct,1)}%) — patrón protector, riesgo CV reducido.`);
  }

  // --- SURGE ---
  if (surgeSbp !== null && surgeSbp > 35) {
    alerts.push({ type:'warn', msg:`⚠️ <strong>Surge matutino marcado</strong> (+${fmt(surgeSbp,1)} mmHg SBP).
      Asociado con mayor riesgo de ictus matutino y eventos coronarios. Posible causa: cobertura insuficiente
      en madrugada. Verificar: ¿el fármaco se toma por la mañana? ¿tiene vida media de 24h?
      Considerar: IECA/ARA-II o CCB de acción ultralarga (telmisartán, amlodipino).
      Confirmar con MAPA 24h si determina cambio terapéutico.` });
  }

  // --- HIPOTENSIÓN EPISÓDICA ---
  if (p.tgtLow > 0) {
    const pctLow = validN > 0 ? (p.tgtLow / validN * 100) : 0;
    if (pctLow > 15) {
      alerts.push({ type:'info', msg:`🔵 ${pctLow.toFixed(0)}% de lecturas con hipotensión (&lt;90/60).
        Evaluar: sobretratamiento, deshidratación, hipotensión ortostática, interacciones. Riesgo de caídas
        especialmente en adulto mayor. Valorar reducción de dosis o simplificación del esquema.` });
    } else {
      points.push(`🔵 Episodios aislados de hipotensión (${p.tgtLow} lectura${p.tgtLow>1?'s':''}).
        Documentar contexto: postprandial, postural, relacionado con medicación o actividad. Vigilar síntomas.`);
    }
  }

  // --- FC ---
  if (overallFc !== null) {
    if (overallFc > 100) {
      points.push(`❤️ <strong>Taquicardia</strong> (FC promedio ${fmt(overallFc,0)} lpm).
        Descartar causas secundarias: anemia, hipertiroidismo, ansiedad, dolor, deshidratación, cafeína.
        En HTA no controlada, la taquicardia refleja activación simpática — el control de la PA suele normalizarla.
        Si el esquema incluye CCB dihidropiridínico (amlodipino), verificar taquicardia refleja por vasodilatación.
        Si persiste FC >100 en reposo: considerar beta-bloqueador (no como antihipertensivo de primera línea,
        pero útil en HTA + taquicardia / HTA + angina / post-IAM).`);
    } else if (overallFc < 60) {
      points.push(`❤️ <strong>Bradicardia</strong> (FC promedio ${fmt(overallFc,0)} lpm).
        Evaluar: beta-bloqueador o CCB no dihidropiridínico (diltiazem/verapamilo) a dosis excesiva,
        hipotiroidismo, bloqueo AV. Si es sintomática (mareo, síncope, fatiga): ajustar dosis o cambiar clase.
        Importante: no combinar beta-bloqueador + diltiazem/verapamilo por riesgo de bloqueo AV.`);
    } else if (overallFc >= 60 && overallFc <= 80) {
      points.push(`❤️ FC promedio ${fmt(overallFc,0)} lpm — rango óptimo.
        En HTA + angina o post-IAM, meta de FC ≤70 lpm con beta-bloqueador.
        En HTA sin comorbilidades CV: FC entre 60–80 lpm en reposo es fisiológica.`);
    } else {
      points.push(`❤️ FC promedio ${fmt(overallFc,0)} lpm — rango normal-alto. Vigilar si aumenta;
        descartar activación simpática por HTA no controlada, ansiedad o efecto de vasodilatador sin beta-bloqueador.`);
    }
    if (sdFc !== null && sdFc > 15) {
      points.push(`❤️ <strong>Alta variabilidad de FC</strong> (DS = ${fmt(sdFc,1)} lpm).
        Puede indicar: arritmia (fibrilación auricular — revisar regularidad del pulso en cada toma),
        medición en diferentes condiciones de esfuerzo o estrés, o fluctuación farmacológica.
        Si hay irregularidad de pulso: ECG.`);
    }
  }

  // --- VARIABILIDAD TA ---
  if (sdSbp !== null && sdSbp > 15) {
    points.push(`📉 <strong>Alta variabilidad tensional</strong> (DS SBP = ${fmt(sdSbp,1)} mmHg).
      La variabilidad visita-a-visita se ha descrito como predictor independiente de eventos CV (Rothwell et al.,
      <em>Lancet</em> 2010). Causas: técnica de medición inconsistente, adherencia irregular, dolor/estrés, AINE,
      esquema de acción corta. Considerar combinación fija de acción prolongada.`);
  }

  // --- VARIACIÓN HORARIA ---
  const mSbp = periodAvg['manana'].sbp, nSbp = periodAvg['noche'].sbp;
  if (mSbp !== null && nSbp !== null && Math.abs(mSbp - nSbp) > 20) {
    points.push(`⏱️ <strong>Variación horaria significativa</strong> (mañana ${fmt(mSbp,0)} vs. noche ${fmt(nSbp,0)} mmHg,
      Δ ${fmt(Math.abs(mSbp-nSbp),0)} mmHg). Revisar vida media del fármaco y horario de administración.
      ¿El esquema cubre las 24h? Evaluar si el pico coincide con momento de mayor riesgo CV (6–10 am).`);
  }

  // --- AJUSTE FARMACOLÓGICO (solo si hay datos suficientes y PA fuera de meta) ──
  const diasConDatos = p.daysWithData || 0;
  const oSbp = overallSbp, oDbp = overallDbp;

  if (diasConDatos >= 5 && validN >= 12 && oSbp !== null && (oSbp >= target.sbp || oDbp >= target.dbp)) {
    const varNote = (sdSbp !== null && sdSbp > 15)
      ? `<div style="margin-top:5px;">• Variabilidad alta: preferir fármacos de acción prolongada (<abbr title="Calcioantagonista">CCB</abbr>/<abbr title="Antagonista del Receptor de Angiotensina II">ARA-II</abbr> de 24h).</div>` : '';
    const eshUmbral = (oSbp >= 135 || oDbp >= 85) ? 'Supera umbral domiciliario ≥135/85.' : 'Bajo umbral domiciliario ESH; evaluar contexto.';

    points.push(`<div style="border-top:1px solid var(--c-border);padding-top:10px;margin-top:10px;">
      <strong style="font-size:12px;color:var(--c-head);">⏱️ Recomendación de Ajuste Farmacológico</strong>
      <div style="margin-top:8px;font-size:11.5px;line-height:1.7;display:grid;gap:6px;">
        <div>• <strong>NOM-030-SSA2:</strong> considerar ajuste si apego ≥4–8 sem verificado y causas modificables descartadas. <abbr title="Guía de Práctica Clínica">GPC</abbr>-CENETEC: <abbr title="Inhibidor de la Enzima Convertidora de Angiotensina">IECA</abbr>/<abbr title="Antagonista del Receptor de Angiotensina II">ARA-II</abbr> + <abbr title="Calcioantagonista">CCB</abbr> en <abbr title="Hipertensión Arterial">HTA</abbr> Grado 2.</div>
        <div>• <strong>AHA/ACC 2025:</strong> PA ≥140/90 → combinación de inicio. PA 130–139/80–89 + riesgo PREVENT ≥7.5% (o <abbr title="Enfermedad Cardiovascular Aterosclerótica">ASCVD</abbr> clínica, DM, ERC) → iniciar/intensificar. Seguimiento 1 mes post-ajuste.</div>
        <div>• <strong>ESH 2023:</strong> ${eshUmbral} Control post-ajuste cada 1–3 meses hasta meta, luego cada 6 meses.</div>
        ${varNote}
      </div>
    </div>`);
  }

  // --- NOTA METODOLÓGICA ---
  alerts.push({ type:'note', msg:`📌 <strong>Limitación:</strong> esta herramienta es de apoyo clínico —
    no sustituye el juicio médico ni equivale a MAPA 24h (estándar de referencia). Los patrones circadianos calculados
    aquí son aproximaciones con lecturas domiciliarias. Siempre confirmar hallazgos críticos (non-dipper,
    surge marcado, variabilidad alta) con MAPA 24h antes de modificar tratamiento.` });

  el.innerHTML = `<ul>${points.map(pt=>'<li>'+pt+'</li>').join('')}</ul>`;
  alEl.innerHTML = alerts.map(a=>`<div class="alert-box alert-${a.type}">${a.msg}</div>`).join('');
}

const STORAGE_AVAILABLE = (() => {
  try {
    const testKey = '__tensiolab_test__';
    localStorage.setItem(testKey, '1');
    localStorage.removeItem(testKey);
    return true;
  } catch (e) { return false; }
})();

function guardar(clave, valor) {
  try { localStorage.setItem(clave, JSON.stringify(valor)); return true; }
  catch (e) { console.warn('localStorage no disponible:', e); return false; }
}
function cargar(clave) {
  try {
    const item = localStorage.getItem(clave);
    return item ? { ok: true, data: JSON.parse(item) } : { ok: false, data: null };
  } catch (e) { return { ok: false, data: null, error: true }; }
}

function saveData() {
  const grid = {};
  const SAVE_DAYS = getDays();
  for (let day = 0; day < SAVE_DAYS; day++) {
    PERIODS.forEach(p => {
      grid[`sbp_${day}_${p}`] = document.getElementById(`sbp_${day}_${p}`)?.value || '';
      grid[`dbp_${day}_${p}`] = document.getElementById(`dbp_${day}_${p}`)?.value || '';
      grid[`fc_${day}_${p}`]  = document.getElementById(`fc_${day}_${p}`)?.value || '';
    });
  }
  const ok = guardar('tensiolab_v2', { days: getDays(), profile: getProfile(), grid });
  alert(ok ? '✅ Datos guardados.' : '❌ No se pudo guardar: el almacenamiento local no está disponible en este navegador/contexto.');
}

function loadData() {
  const result = cargar('tensiolab_v2');
  if (result.error) { alert('❌ El almacenamiento local no está disponible en este navegador/contexto.'); return; }
  if (!result.ok) { alert('No hay datos guardados.'); return; }
  const data = result.data;
  const daysEl = document.getElementById('cfg-days');
  const profEl = document.getElementById('cfg-profile');
  if (daysEl && data.days) { daysEl.value = data.days; buildTable(); }
  if (profEl && data.profile) profEl.value = data.profile;
  if (data.grid) {
    for (let day = 0; day < getDays(); day++) {
      PERIODS.forEach(p => {
        const sbpEl = document.getElementById(`sbp_${day}_${p}`);
        const dbpEl = document.getElementById(`dbp_${day}_${p}`);
        const fcEl  = document.getElementById(`fc_${day}_${p}`);
        if (sbpEl) sbpEl.value = data.grid[`sbp_${day}_${p}`] || '';
        if (dbpEl) dbpEl.value = data.grid[`dbp_${day}_${p}`] || '';
        if (fcEl ) fcEl.value  = data.grid[`fc_${day}_${p}`]  || '';
      });
    }
    document.querySelectorAll('.bp-input').forEach(el => el.dispatchEvent(new Event('input')));
  }
  alert('✅ Datos cargados.');
}

function clearData() {
  if (!confirm('¿Limpiar todos los datos? Esta acción no se puede deshacer.')) return;
  const DAYS = getDays();
  for (let day = 0; day < DAYS; day++) {
    PERIODS.forEach(p => {
      const sbpEl = document.getElementById(`sbp_${day}_${p}`);
      const dbpEl = document.getElementById(`dbp_${day}_${p}`);
      const fcEl  = document.getElementById(`fc_${day}_${p}`);
      const pamEl = document.getElementById(`pam_${day}_${p}`);
      if (sbpEl) { sbpEl.value = ''; sbpEl.className = 'bp-input'; }
      if (dbpEl) { dbpEl.value = ''; dbpEl.className = 'bp-input'; }
      if (fcEl)  fcEl.value = '';
      if (pamEl) pamEl.textContent = '—';
    });
    ['sbp','dbp','fc','pam'].forEach(type => {
      const avgEl = document.getElementById(`avg_${type}_${day}`);
      if (avgEl) avgEl.textContent = '—';
    });
  }
  document.getElementById('results-section').style.display = 'none';
}
